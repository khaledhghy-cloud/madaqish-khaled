صاحب أعلى مزايدة هو اللاعب " + (highestBidder + 1) + " بمبلغ " + highestBid); } update(); } function advanceAuction() { actionsCount++; if (actionsCount >= 6) { finishAuction(); return; } turn = (starter + actionsCount) % 6; message("دور اللاعب " + (turn + 1)); update(); } function raiseBid() { if (!dealt) { message("وزع الأوراق أولاً"); return; } let amount = prompt("أدخل مبلغ المزايدة، مثل 5 أو 50 أو 90:"); if (amount === null) return; amount = Number(amount); if (!Number.isInteger(amount) || amount < 5 || amount % 5 !== 0) { message("المزايدة تبدأ من 5 وتكون بالخمسات"); return; } if (amount <= highestBid) { message("يجب أن تكون المزايدة أعلى من أعلى مبلغ حالي"); return; } if (amount > players[turn].score) { message("المبلغ أكبر من رصيد اللاعب"); return; } highestBid = amount; highestBidder = turn; pot = amount; active.push(turn); message("اللاعب " + (turn + 1) + " زايد بمبلغ " + amount); advanceAuction(); } function passBid() { if (!dealt) { message("وزع الأوراق أولاً"); return; } passed.push(turn); message("اللاعب " + (turn + 1) + " انسحب من المزايدة"); advanceAuction(); } function marketEntry() { if (!dealt) { message("وزع الأوراق أولاً"); return; } if (actionsCount >= 6) { message("انتهت المزايدة"); return; } marketPlayers.push(turn); active.push(turn); message("اللاعب " + (turn + 1) + " قال سوقي"); advanceAuction(); } function nextRound() { round++; hands = []; dealt = false; turn = (round - 1) % 6; starter = turn; actionsCount = 0; highestBid = 0; highestBidder = -1; pot = 0; active = []; passed = []; marketPlayers = []; message("الجولة " + round + ". البداية من اللاعب " + (starter + 1)); update(); } function resetGame() { round = 1; players.forEach(p => p.score = 100); nextRound(); round = 1; turn = 0; starter = 0; update(); }
